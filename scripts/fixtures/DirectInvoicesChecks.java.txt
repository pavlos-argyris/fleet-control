/** Executes the actual Direct scheduler with disposable native/network ports. */
public class DirectInvoicesChecks {
 static void check(boolean b,String message){if(!b)throw new AssertionError(message);}
 static void pulse(IndependentInvoices worker)throws Exception{java.lang.reflect.Field f=IndependentInvoices.class.getDeclaredField("next");f.setAccessible(true);f.setLong(worker,0);worker.tick();}
 static void until(java.util.concurrent.Callable<Boolean> condition)throws Exception{long end=System.currentTimeMillis()+10000;while(!condition.call()){if(System.currentTimeMillis()>end)throw new AssertionError("scheduler deadline");Thread.sleep(5);}}
 static LocalWorkQueue queue(IndependentInvoices worker)throws Exception{java.lang.reflect.Field f=IndependentInvoices.class.getDeclaredField("queue");f.setAccessible(true);return(LocalWorkQueue)f.get(worker);}
 public static void main(String[] args)throws Exception{
  android.content.Context c=new android.content.Context();IndependentRuntime runtime=new IndependentRuntime();FleetDataStore.disk="{}";
  for(int i=0;i<62;i++)runtime.add("job-"+i,false);IndependentInvoices worker=new IndependentInvoices(c,runtime);
  InvoicePipeline.entered=new CountDownLatch(2);InvoicePipeline.release=new CountDownLatch(1);pulse(worker);check(InvoicePipeline.entered.await(5,TimeUnit.SECONDS),"two actual scheduler workers enter concurrently");check(InvoicePipeline.maximum.get()==2,"bounded two-job lane");InvoicePipeline.release.countDown();
  while(runtime.complete.get()<62){pulse(worker);Thread.sleep(10);}check(runtime.analyses.get()==62,"62 exactly-once local analysis results");worker.close();until(()->InvoicePipeline.active.get()==0);
  FleetDataStore.disk="{}";runtime=new IndependentRuntime();runtime.add("cancel-race",false);worker=new IndependentInvoices(c,runtime);InvoicePipeline.entered=new CountDownLatch(1);InvoicePipeline.release=new CountDownLatch(1);pulse(worker);check(InvoicePipeline.entered.await(5,TimeUnit.SECONDS),"cancellation starts after read");runtime.cancel("cancel-race");InvoicePipeline.release.countDown();until(()->InvoicePipeline.active.get()==0);Thread.sleep(30);check(runtime.analyses.get()==0,"cancelled job cannot attach its late result");worker.close();
  FleetDataStore.disk="{}";runtime=new IndependentRuntime();runtime.add("offline-qr",true);DeviceHealth.online=false;worker=new IndependentInvoices(c,runtime);pulse(worker);check(runtime.analyses.get()==0,"offline QR does not start");JSONObject row=queue(worker).ready(Long.MAX_VALUE).getJSONObject(0);check(row.optInt("directAttempts")==0,"offline consumes no attempts");DeviceHealth.online=true;queue(worker).update("offline-qr","retryAt",0);pulse(worker);final IndependentRuntime onlineRuntime=runtime;until(()->onlineRuntime.complete.get()==1);worker.close();
  FleetDataStore.disk="{}";runtime=new IndependentRuntime();runtime.add("failing",false);worker=new IndependentInvoices(c,runtime);InvoicePipeline.fail=true;InvoicePipeline.release=new CountDownLatch(0);
  for(int attempt=1;attempt<=6;attempt++){queue(worker).update("failing","retryAt",0);pulse(worker);final int expected=attempt;final IndependentRuntime failingRuntime=runtime;until(()->failingRuntime.errors.get()>=expected);until(()->InvoicePipeline.active.get()==0);Thread.sleep(15);if(attempt==3){worker.close();worker=new IndependentInvoices(c,runtime);}}
  check(runtime.errors.get()==6&&runtime.status("failing").equals("error"),"six attempts total across restart, final state persists");pulse(worker);check(runtime.errors.get()==6,"terminal job cannot restart automatically");worker.close();InvoicePipeline.fail=false;
  System.out.println("PASS actual Direct scheduler:62 jobs, two bounded workers, cancellation before commit, offline zero attempts, retry exhaustion across restart. Synthetic native ports only.");
 }
}

class IndependentRuntime {
 final Map<String,JSONObject> jobs=new LinkedHashMap<String,JSONObject>();final AtomicInteger complete=new AtomicInteger(),analyses=new AtomicInteger(),errors=new AtomicInteger();
 synchronized void add(String id,boolean qr){JSONObject payload=new JSONObject();if(qr)payload.put("qrUrl","https://fixture.invalid/invoice").put("bikeId","bike").put("entryId",id);else payload.put("documentKey","service-documents/"+id+"/invoice.pdf");jobs.put(id,new JSONObject().put("id",id).put("type","service_document_render").put("status","pending").put("payload",payload));}
 synchronized void cancel(String id){jobs.get(id).put("status","cancelled");}synchronized String status(String id){return jobs.get(id).optString("status");}String realm(){return"https://fixture.invalid";}
 synchronized JSONObject internal(String path,String method,JSONObject body)throws Exception{
  if(path.startsWith("/api/server/pending")){java.net.URI uri=new java.net.URI(path);Map<String,String> query=new HashMap<String,String>();for(String pair:uri.getRawQuery().split("&")){String[] p=pair.split("=",2);query.put(p[0],p.length==2?java.net.URLDecoder.decode(p[1],"UTF-8"):"");}JSONArray ops=new JSONArray(),known=new JSONArray();int limit=Integer.parseInt(query.get("limit"));for(JSONObject job:jobs.values())if(job.optString("status").equals("pending")&&ops.length()<limit)ops.put(new JSONObject(job.toString()));JSONArray ids=new JSONArray(query.get("known"));for(int i=0;i<ids.length();i++){String id=ids.getString(i);known.put(new JSONObject().put("id",id).put("status",jobs.containsKey(id)?status(id):"missing"));}return new JSONObject().put("operations",ops).put("known",known);}
  if(path.endsWith("analysis-result")){analyses.incrementAndGet();return new JSONObject();}
  if(path.endsWith("/ack")){JSONArray ids=body.getJSONArray("ids");for(int i=0;i<ids.length();i++)if(status(ids.getString(i)).equals("pending")){jobs.get(ids.getString(i)).put("status","applied");complete.incrementAndGet();}return new JSONObject();}
  if(path.endsWith("operation-error")){int n=body.getInt("directAttemptCount");jobs.get(body.getString("jobId")).getJSONObject("payload").put("directAttemptCount",n);if(body.optBoolean("terminal")||n>=6)jobs.get(body.getString("jobId")).put("status","error");errors.incrementAndGet();return new JSONObject();}
  return new JSONObject();
 }
