package gr.argyrismitsos.fleetserver;
import android.content.Context;import android.database.sqlite.SQLiteDatabase;
import org.json.*;import java.io.*;import java.nio.file.*;import java.util.*;

public final class BusinessBundleChecks {
    static void check(boolean value,String message){if(!value)throw new AssertionError(message);}
    static JSONObject json(File file)throws Exception{return new JSONObject(new String(Files.readAllBytes(file.toPath()),"UTF-8"));}
    public static void run(Context context,SQLiteDatabase db,BusinessSqlite sql)throws Exception{
        File directory=new File(context.getFilesDir(),"safe-backups"),data=new File(context.getCacheDir(),"read-data.json"),database=new File(context.getCacheDir(),"read-database.sqlite"),documents=new File(context.getCacheDir(),"read-documents.zip");
        JSONObject recovery=new JSONObject().put("schemaVersion",1).put("realm","https://fleet.fixture.invalid").put("authPepper",BusinessTest.PEPPER).put("recoveryToken","rrrrrrrrrrrrrrrrrrrrrrrrrrrrrrrrrrrrrrrrrrr").put("ownerId","synthetic-owner").put("sourceGeneration","11111111-1111-4111-8111-111111111111").put("sourceFingerprint","a".repeat(64)).put("privateKey","MUST_NOT_BACK_UP").put("ordexaPassword","MUST_NOT_BACK_UP");
        String id=UUID.randomUUID().toString();List<String> stages=new ArrayList<String>();
        try(BackupEnvelope envelope=new BackupEnvelope(PortableBackupCipher.create("Συνθετικός φορητός κωδικός 2026!"),new byte[32])){
            JSONObject parts=NativeBackupBuilder.prepare(context,db,directory,id,new JSONObject().put("workerUrl","https://fleet.fixture.invalid"),recovery,envelope,(stage,done,total)->stages.add(stage));
            check(parts.getBoolean("localReady")&&parts.getBoolean("portable")&&parts.getJSONObject("nativeProof").getBoolean("restoreVerified")&&parts.getJSONObject("nativeProof").getBoolean("documentsVerified"),"complete encrypted bundle includes actual decrypted SQLite/document restore proof");
            for(String prefix:new String[]{"data","database","docs"}){File file=new File(directory,parts.getString(prefix+"Path"));check(file.length()==parts.getLong(prefix+"Bytes")&&BackupCipher.digest(file).equals(parts.getString(prefix+"Sha256")),"all three immutable encrypted parts have exact hashes");}
            envelope.decrypt(new FileInputStream(new File(directory,parts.getString("dataPath"))),data,"data",null);JSONObject root=json(data);check(root.getInt("schemaVersion")==6&&root.has("nativeDatabase")&&!root.has("fleet"),"new format cannot be mistaken for a legacy empty fleet");check(!root.getJSONObject("nativeRecovery").has("privateKey")&&!root.toString().contains("MUST_NOT_BACK_UP"),"Keystore identity and provider credentials excluded by allowlist");
            envelope.decrypt(new FileInputStream(new File(directory,parts.getString("databasePath"))),database,"database",null);envelope.decrypt(new FileInputStream(new File(directory,parts.getString("docsPath"))),documents,"documents",null);
            check(NativeBackupRestoreProbe.run(context,database,root.getJSONObject("nativeDatabase"),documents).getBoolean("restoreVerified"),"exported encrypted bytes independently reconstruct the full snapshot and exact original files");
            root.put("_portableAuthenticated",true);
            NativeBackupReader.Source source=kind->new FileInputStream(new File(directory,parts.getString(kind.equals("database")?"databasePath":"docsPath")));
            NativeBackupReader.Decoder decoder=(input,file,kind)->envelope.decrypt(input,file,kind,null);
            try(NativeBackupReader.Prepared prepared=NativeBackupReader.open(root,context.getCacheDir(),source,decoder)){
                check(prepared.verification.getBoolean("complete")&&!prepared.verification.getBoolean("restoreVerified"),"reading/CRC/hash checking a bundle never claims an isolated restore");
                check(NativeBackupRestoreProbe.run(context,prepared.database,prepared.manifest,prepared.documents).getBoolean("restoreVerified"),"exact reader outputs support independent full recovery");
            }
            try{NativeBackupReader.validate(root,false);throw new AssertionError("unauthenticated native manifest accepted");}catch(IOException expected){}
            JSONObject mismatched=new JSONObject(root.toString());mismatched.getJSONObject("nativeDatabase").put("encryptedSha256","0".repeat(64));
            try(NativeBackupReader.Prepared ignored=NativeBackupReader.open(mismatched,context.getCacheDir(),source,decoder)){throw new AssertionError("mixed database companion accepted");}catch(IOException expected){}
            try(NativeBackupReader.Prepared ignored=NativeBackupReader.open(root,context.getCacheDir(),kind->null,decoder)){throw new AssertionError("missing database accepted");}catch(IOException expected){}
            mismatched=new JSONObject(root.toString());mismatched.getJSONObject("nativeDatabase").getJSONArray("objects").getJSONObject(0).put("name","Wrong invoice.pdf");
            try(NativeBackupReader.Prepared ignored=NativeBackupReader.open(mismatched,context.getCacheDir(),source,decoder)){throw new AssertionError("objects different from DB accepted");}catch(IOException expected){}
            for(File child:context.getCacheDir().listFiles())check(!child.getName().startsWith("native-backup-read-"),"private reader plaintext cleaned on success and every failure");
            JSONObject second=NativeBackupBuilder.prepare(context,db,directory,UUID.randomUUID().toString(),new JSONObject().put("workerUrl","https://fleet.fixture.invalid"),recovery,envelope,false,(stage,done,total)->{});
            check(second.getBoolean("documentsReused")&&second.getString("docsPath").equals(parts.getString("docsPath")),"unchanged invoice archive/encrypted companion reused across different DB snapshots");
            check(!second.getJSONObject("nativeProof").getBoolean("restoreVerified")&&second.getJSONObject("nativeProof").getBoolean("integrityVerified"),"fast ordinary backup does not falsely claim a new isolated restore");
        }finally{data.delete();database.delete();documents.delete();}
        check(stages.stream().anyMatch(s->s.contains("δοκιμή ανάκτησης")),"actual isolated restore stage visible in progress");
        check(sql.call("sql.query",new JSONObject().put("sql","SELECT COUNT(*) n FROM native_backup_pins")).getJSONArray("results").getJSONObject(0).getInt("n")==0,"completed bundle leaves no durable object pins");
        System.out.println("PASS full native encrypted backup bundle: three actual FCB3 parts, complete SQLite and private documents, no device/provider secrets, independent decrypt/real isolated restore, immutable hashes, cached PDF companion and honest fast-backup versus restore proof.");
    }
}
