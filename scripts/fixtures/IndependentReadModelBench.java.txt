  }

   try{check(running.await(5,TimeUnit.SECONDS),"benchmark writer acquired real SQLite transaction");List<Future<?>> work=new ArrayList<Future<?>>();for(int i=0;i<clients;i++)work.add(pool.submit(()->{try{long start=System.nanoTime();JSONObject response=cached?models.get(request):queuedRead(q);check(response!=null&&value(response).getJSONArray("couriers").getJSONObject(0).getInt("deliveryDueCents")==0,"identical committed amount while the writer is busy");times.add((System.nanoTime()-start)/1000000.0);}catch(Exception error){throw new RuntimeException(error);}}));for(Future<?> task:work)task.get(10,TimeUnit.SECONDS);if(cached)check(!busy.isDone(),"cached reads finish before the heavy writer releases its transaction");release.countDown();busy.get(10,TimeUnit.SECONDS);
    List<Double> sorted=new ArrayList<Double>(times);Collections.sort(sorted);JSONObject report=new JSONObject().put("clients",clients).put("cached",cached).put("requests",times.size());for(int n:new int[]{50,95,99})report.put("p"+n+"Ms",sorted.get(Math.min(sorted.size()-1,(int)Math.ceil(sorted.size()*n/100.0)-1)));return report;
   }finally{release.countDown();pool.shutdownNow();q.shutdownNow();}
  }
