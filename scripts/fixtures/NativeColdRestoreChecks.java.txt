package gr.argyrismitsos.fleetserver;
import org.json.*;import java.io.*;import java.nio.file.*;import java.nio.channels.FileChannel;import java.util.*;

/** Real filesystem process-death/rollback checks with the actual authenticated SQLite backup. */
public final class NativeColdRestoreChecks {
    static class Crash extends Error{}
    static void check(boolean value,String message){if(!value)throw new AssertionError(message);}
    static void write(File file,String value)throws Exception{file.getParentFile().mkdirs();try(FileOutputStream out=new FileOutputStream(file)){out.write(value.getBytes("UTF-8"));out.getFD().sync();}}
    static String text(File file)throws Exception{return new String(Files.readAllBytes(file.toPath()),"UTF-8");}
    static void copy(File source,File target)throws Exception{target.getParentFile().mkdirs();Files.copy(source.toPath(),target.toPath(),StandardCopyOption.REPLACE_EXISTING);try(FileOutputStream sync=new FileOutputStream(target,true)){sync.getFD().sync();}}
    public static void run(File base,File backupDatabase,JSONObject manifest,File sourceDocuments,String owner)throws Exception{
        List<String> checkpoints=Arrays.asList("swapping","move:databases","move:new-databases","database_installed","move:service-documents","move:new-documents","documents_installed","installing_secrets","secrets-installed","installed");
        for(boolean crash:new boolean[]{true,false})for(String failure:checkpoints){
            File app=new File(base,(crash?"crash-":"failure-")+failure.replace(':','_')),files=new File(app,"files"),databaseDirectory=new File(app,"databases"),documents=new File(files,"service-documents"),vault=new File(files,"fixture-vault");String id=UUID.randomUUID().toString();
            write(new File(databaseDirectory,NativeColdRestore.DATABASE_NAME),"original database");write(new File(databaseDirectory,NativeColdRestore.DATABASE_NAME+"-wal"),"original committed WAL");write(new File(databaseDirectory,NativeColdRestore.DATABASE_NAME+"-shm"),"original shared memory");write(new File(documents,"original-private.pdf"),"original document");write(vault,"original identity");
            final JSONObject identity=new JSONObject().put("realm",NativeMigrationFlowTest.REALM).put("generation","77777777-1111-4111-8111-111111111111").put("ownerId",owner);final String[] point={""};final boolean[] injected={false};
            NativeColdRestore.Ports ports=new NativeColdRestore.Ports(){
                public void verifyPending(JSONObject marker)throws Exception{check(marker.getString("id").equals(id)&&marker.getString("pendingHash").equals("a".repeat(64)),"journal is bound to private pending identity");}
                public void verifySource(File db,JSONObject expected)throws Exception{NativeDatabaseSnapshot.verifyReadOnly(db,expected);}
                public void installSecrets(JSONObject marker)throws Exception{NativeRecoveryInstall.prepare(new File(databaseDirectory,NativeColdRestore.DATABASE_NAME),documents,manifest,identity);write(vault,"new identity");}
                public void rollbackSecrets(JSONObject marker)throws Exception{write(vault,"original identity");}
                public void syncDirectory(File directory)throws Exception{try(FileChannel channel=FileChannel.open(directory.toPath(),StandardOpenOption.READ)){channel.force(true);}}
                public void checkpoint(String name)throws Exception{if(name.equals(point[0])&&!injected[0]){injected[0]=true;if(crash)throw new Crash();throw new IOException("synthetic I/O interruption at "+name);}}
            };
            NativeColdRestore restore=new NativeColdRestore(files,databaseDirectory,documents,ports);File staged=restore.directory(id),newDb=new File(staged,"new-databases"),newDocs=new File(staged,"new-documents");newDocs.mkdirs();copy(backupDatabase,new File(newDb,NativeColdRestore.DATABASE_NAME));JSONArray objects=manifest.getJSONArray("objects");for(int i=0;i<objects.length();i++){String name=objects.getJSONObject(i).getString("local_name");copy(new File(sourceDocuments,name),new File(newDocs,name));}
            restore.arm(id,manifest,"a".repeat(64));check(text(new File(databaseDirectory,NativeColdRestore.DATABASE_NAME)).equals("original database"),"arming never modifies live SQLite");point[0]=failure;
            try{restore.apply();throw new AssertionError("fault not injected "+failure);}catch(Crash expected){check(crash,"only simulated process death");}catch(IOException expected){check(!crash,"only simulated ordinary I/O failure: "+expected);}
            check(injected[0],"each critical step was exercised "+failure);point[0]="";
            if(crash){new NativeColdRestore(files,databaseDirectory,documents,ports).apply();check(restore.state().getString("phase").equals("installed"),"fresh process completes exactly once after "+failure);check(text(vault).equals("new identity"),"new identity accompanies new database and documents");NativeDatabaseSnapshot.verifyRecovering(new File(databaseDirectory,NativeColdRestore.DATABASE_NAME),manifest,identity.getString("realm"),identity.getString("generation"));check(text(new File(staged,"old-databases/"+NativeColdRestore.DATABASE_NAME+"-wal")).equals("original committed WAL"),"original main/WAL/SHM group retained together");check(new File(staged,"old-documents/original-private.pdf").isFile(),"original documents retained for rollback");check(!restore.apply(),"completed journal is idempotent");}
            else{check(restore.state().getString("phase").equals("failed"),"ordinary failure records a safe rollback "+failure);check(text(new File(databaseDirectory,NativeColdRestore.DATABASE_NAME)).equals("original database")&&text(new File(databaseDirectory,NativeColdRestore.DATABASE_NAME+"-wal")).equals("original committed WAL")&&text(new File(documents,"original-private.pdf")).equals("original document")&&text(vault).equals("original identity"),"rollback restores complete original database/WAL/documents/identity group after "+failure);check(!restore.apply(),"failed restore never silently retries on startup");}
            String journal=text(new File(files,"native-recovery/state.json"));check(!journal.contains("recoveryToken")&&!journal.contains("authPepper")&&!journal.contains("privateKey"),"recovery capabilities never enter ordinary journal");
        }
        // A corrupt source must be rejected before the live directory can move.
        File app=new File(base,"corrupt"),files=new File(app,"files"),live=new File(app,"databases"),docs=new File(files,"service-documents");String id=UUID.randomUUID().toString();write(new File(live,NativeColdRestore.DATABASE_NAME),"original");
        NativeColdRestore restore=new NativeColdRestore(files,live,docs,new NativeColdRestore.Ports(){public void verifyPending(JSONObject m){}public void verifySource(File db,JSONObject m)throws Exception{NativeDatabaseSnapshot.verifyReadOnly(db,m);}public void installSecrets(JSONObject m){throw new AssertionError("corrupt source installed");}public void rollbackSecrets(JSONObject m){}public void syncDirectory(File f){}});File staged=restore.directory(id);new File(staged,"new-documents").mkdirs();write(new File(staged,"new-databases/"+NativeColdRestore.DATABASE_NAME),"tampered");boolean rejected=false;try{restore.arm(id,manifest,"a".repeat(64));}catch(IOException expected){rejected=true;}check(rejected&&text(new File(live,NativeColdRestore.DATABASE_NAME)).equals("original"),"corrupt backup cannot arm production replacement");rejected=false;try{restore.directory("../outside");}catch(IOException expected){rejected=true;}check(rejected,"no caller-chosen restore path");
        System.out.println("PASS actual cold restore: authenticated real SQLite/private files, process death and ordinary I/O rollback at10 journal/move/identity boundaries, complete original main/WAL/SHM/doc preservation, current recovering-only state, idempotent startup, corrupt-source/path rejection and no recovery secrets in journal. Android Keystore/fsync adapter remains a physical-device check.");
    }
}
