package gr.argyrismitsos.fleetserver;
import android.database.sqlite.*;import org.json.*;import java.io.*;

public final class BusinessSnapshotChecks {
    static void check(boolean value,String message){if(!value)throw new AssertionError(message);}
    static long count(BusinessSqlite sql,String table)throws Exception{return sql.call("sql.query",new JSONObject().put("sql","SELECT COUNT(*) n FROM "+table)).getJSONArray("results").getJSONObject(0).getLong("n");}
    public static void run(SQLiteDatabase db,BusinessSqlite sql,File directory)throws Exception{
        db.execSQL(NativeDatabaseSnapshot.STATE_SCHEMA);
        db.execSQL("INSERT INTO native_primary_state(id,phase,realm,generation,owner_id) VALUES(1,'active','https://fleet.fixture.invalid','11111111-1111-4111-8111-111111111111','synthetic-owner')");
        long sessions=count(sql,"auth_sessions"),cash=count(sql,"native_cash_events"),receipts=count(sql,"native_operation_receipts"),reviews=count(sql,"native_payroll_reviews");
        NativeDatabaseSnapshot.Capture captured=NativeDatabaseSnapshot.create(db,directory);check(captured.file.length()>5*1024*1024,"real SQLite snapshot includes >5 MiB historical cell without JSON materialization");
        check(count(sql,"auth_sessions")==sessions&&count(sql,"native_cash_events")==cash&&count(sql,"native_operation_receipts")==receipts,"live sessions and business rows are unchanged by snapshot sanitation");
        SQLiteDatabase target=SQLiteDatabase.openDatabase(captured.file.getAbsolutePath(),null,SQLiteDatabase.OPEN_READWRITE);try{
            BusinessSqlite copy=new BusinessSqlite(target);check(count(copy,"auth_sessions")==0&&count(copy,"native_file_transfers")==0,"portable snapshot excludes live sessions and private handles");
            check(count(copy,"native_cash_events")==cash&&count(copy,"native_operation_receipts")==receipts&&count(copy,"native_payroll_reviews")==reviews,"cash, receipts and versioned review audit all survive in exact typed SQLite");
            JSONObject state=copy.call("sql.query",new JSONObject().put("sql","SELECT phase,generation FROM native_primary_state WHERE id=1")).getJSONArray("results").getJSONObject(0);check(state.getString("phase").equals("needs_recovery")&&state.getString("generation").isEmpty(),"restored DB never reactivates the previous Server identity");
        }finally{target.close();}
        JSONObject proof=NativeDatabaseSnapshot.verifyIsolated(captured.file,captured.manifest,new File(directory,"probe"));check(proof.getBoolean("restoreVerified")&&proof.getBoolean("isolated"),"actual independent SQLite restore, schema/row checks, read/write/rollback and integrity proof");
        check(count(sql,"auth_sessions")==sessions&&count(sql,"native_payroll_reviews")==reviews,"isolated restore never touches the live database");
        check(count(sql,"native_backup_pins")>=1,"immutable document versions pinned until archive preparation completes");NativeDatabaseSnapshot.release(db,captured.id);check(count(sql,"native_backup_pins")==0,"archive pins released explicitly after checkpoint");
        try(RandomAccessFile file=new RandomAccessFile(captured.file,"rw")){file.seek(128);file.write(0xff);}boolean rejected=false;try{NativeDatabaseSnapshot.verifyIsolated(captured.file,captured.manifest,new File(directory,"probe"));}catch(IOException expected){rejected=true;}check(rejected,"corrupt SQLite rejected before isolated open or production restore");
        System.out.println("PASS real full SQLite snapshot/isolated recovery: VACUUM including large history, complete cash/receipt/review schema, source unchanged, session/device fencing, exact hashes/inventory, actual write/rollback, document lifecycle pins and corruption rejection.");
    }
}
